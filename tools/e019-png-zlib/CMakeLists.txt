cmake_minimum_required(VERSION 3.22)
project(e019_png_zlib C)
if(NOT ANDROID OR NOT ANDROID_PLATFORM STREQUAL "android-29")
  message(FATAL_ERROR "Android API 29 toolchain required")
endif()
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED ON CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC OFF CACHE BOOL "" FORCE)
set(HAVE___ATTR__VIS_HIDDEN OFF CACHE BOOL "" FORCE)
add_subdirectory("${UPSTREAM}/zlib-1.4" zlib)
# Android uses unversioned libz.so; preserve legacy dlsym consumers.
set_property(TARGET zlib PROPERTY VERSION)
set_property(TARGET zlib PROPERTY SOVERSION)
set_target_properties(zlib PROPERTIES LINK_FLAGS "")
set(PNG_SHARED ON CACHE BOOL "" FORCE)
set(PNG_STATIC OFF CACHE BOOL "" FORCE)
set(PNG_TESTS OFF CACHE BOOL "" FORCE)
set(PNG_TOOLS OFF CACHE BOOL "" FORCE)
set(SKIP_INSTALL_ALL ON CACHE BOOL "" FORCE)
set(PNG_ARM_NEON on CACHE STRING "" FORCE)
set(HAVE_LD_VERSION_SCRIPT OFF CACHE BOOL "" FORCE)
set(HAVE_SOLARIS_LD_VERSION_SCRIPT OFF CACHE BOOL "" FORCE)
set(ZLIB_INCLUDE_DIR "${CMAKE_BINARY_DIR}/zlib;${UPSTREAM}/zlib-1.4" CACHE STRING "" FORCE)
set(ZLIB_LIBRARY zlib CACHE STRING "" FORCE)
add_subdirectory("${UPSTREAM}/libpng-1.6.59" png)
set_property(TARGET png_shared PROPERTY VERSION)
set_property(TARGET png_shared PROPERTY SOVERSION)
set_target_properties(png_shared PROPERTIES OUTPUT_NAME png LINK_FLAGS "")
if(ANDROID_ABI STREQUAL "armeabi-v7a")
  add_library(aeabi_bridge OBJECT aeabi.c)
  set_target_properties(aeabi_bridge PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_compile_options(aeabi_bridge PRIVATE -fno-builtin)
  target_sources(zlib PRIVATE $<TARGET_OBJECTS:aeabi_bridge>)
  target_sources(png_shared PRIVATE $<TARGET_OBJECTS:aeabi_bridge>)
endif()
add_executable(probe probe.c)
target_include_directories(probe PRIVATE "${UPSTREAM}/zlib-1.4" "${CMAKE_BINARY_DIR}/zlib" "${UPSTREAM}/libpng-1.6.59" "${CMAKE_BINARY_DIR}/png")
target_link_libraries(probe PRIVATE dl)
